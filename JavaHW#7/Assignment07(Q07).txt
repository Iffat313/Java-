interface Colorable {
    //declared abstract method, will be overriden by the class that implements it
	public abstract void howToColor();
}

class Sqaure extends GeometricObject implements Colorable
{
	private double Side;

	public Sqaure() {
		Side = 4;
	}

	public void setSide(int x) {
		Side = x;
	}

	public double getSide() {
		return Side;
	}

	@Override
	public void howToColor() {
		System.out.println("Color all 4 sides");
	}

	@Override
	public double getArea() {
		return Side*Side;
	}

	@Override
	public double getPerimeter() {
		return 4*Side;
	}


}

interface Comparable<E> {
    //though the function signature is not abstract, we still override this method
    //it's true that all abstract methods will be overriden, however this does not mean 
    //that overriden methods are exlcusive to those that are abstarct. All sorts of methods can be overriden
    public int compareTo(E o);
}

class ComparableSqaure extends Sqaure implements Comparable<Sqaure>
{
    public ComparableSqaure(double y){
        super(y); //invokes the parameterized superclass constructor of this subclass explicitly
    }
    @Override
    public int compareTo(Sqaure o){
        if(this.getArea() < o.getArea()){
            return -1;
        }
        
        else if(this.getArea() == o.getArea()){
            return 0;
        }
        
        else{
            return 1;
        }
    }
    
    
}
